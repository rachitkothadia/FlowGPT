cmake_minimum_required(VERSION 3.20)
project(FlowGPTCoreTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_executable(FlowGPTCoreTests Tests/CoreTests.cpp)
target_link_libraries(FlowGPTCoreTests PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(FlowGPTCoreTests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(FlowGPTCoreTests PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME CorePolicies COMMAND FlowGPTCoreTests)
